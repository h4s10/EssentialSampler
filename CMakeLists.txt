cmake_minimum_required(VERSION 3.22)
project(EssentialSampler VERSION 0.7.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use this build for Windows x64; use scripts/build.sh on macOS.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0601)
if(MSVC)
    add_compile_options(/utf-8 /W3 /EHsc /bigobj)
endif()
add_library(sampler_core STATIC src/Engine.cpp src/Effects.cpp src/PitchShifter.cpp)
target_include_directories(sampler_core PUBLIC src third_party)
add_library(sampler_editor STATIC src/WindowsEditor.cpp src/AudioFileWindows.cpp)
target_link_libraries(sampler_editor PUBLIC sampler_core user32 gdi32 comdlg32 shell32)
add_library(sampler_vst3 MODULE src/PluginWindows.cpp)
target_link_libraries(sampler_vst3 PRIVATE sampler_editor)
set_target_properties(sampler_vst3 PROPERTIES PREFIX "" OUTPUT_NAME "Essential Sampler" SUFFIX ".vst3")
add_executable(sampler_standalone WIN32 src/WindowsStandalone.cpp)
target_link_libraries(sampler_standalone PRIVATE sampler_editor winmm)
set_target_properties(sampler_standalone PROPERTIES OUTPUT_NAME "Essential Sampler")
add_executable(pitch_benchmark tests/pitch_benchmark.cpp)
target_link_libraries(pitch_benchmark PRIVATE sampler_core)
include(CTest)
if(BUILD_TESTING)
    foreach(suite IN ITEMS engine effects bandwidth fft audio_file windows_host windows_ui)
        add_executable(${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(${suite}_tests PRIVATE sampler_core)
        if(MSVC)
            # Test assertions must remain active in a Release configuration.
            target_compile_options(${suite}_tests PRIVATE /UNDEBUG)
        endif()
    endforeach()
    target_sources(audio_file_tests PRIVATE src/AudioFileWindows.cpp)
    target_link_libraries(windows_ui_tests PRIVATE sampler_editor)
    foreach(suite IN ITEMS engine effects bandwidth fft audio_file windows_ui)
        add_test(NAME ${suite} COMMAND ${suite}_tests)
        set_tests_properties(${suite} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 300)
    endforeach()
    add_test(NAME windows_host COMMAND windows_host_tests "$<TARGET_FILE:sampler_vst3>")
    set_tests_properties(windows_host PROPERTIES TIMEOUT 180)
endif()
